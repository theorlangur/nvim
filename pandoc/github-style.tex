\documentclass[11pt]{article}

% Page geometry
\usepackage{geometry}
\geometry{letterpaper, margin=1.2in}

% Typography (Cross-platform XeLaTeX fonts via filenames)
\usepackage{fontspec}

\setmainfont{texgyreheros}[
  Extension = .otf,
  UprightFont = *-regular,
  BoldFont = *-bold,
  ItalicFont = *-italic,
  BoldItalicFont = *-bolditalic
]

\setsansfont{texgyreheros}[
  Extension = .otf,
  UprightFont = *-regular,
  BoldFont = *-bold,
  ItalicFont = *-italic,
  BoldItalicFont = *-bolditalic
]
\renewcommand{\familydefault}{\sfdefault} % Makes the whole document sans-serif

\setmonofont{texgyrecursor}[
  Scale=0.9,
  Extension = .otf,
  UprightFont = *-regular,
  BoldFont = *-bold,
  ItalicFont = *-italic,
  BoldItalicFont = *-bolditalic
]

% Paragraph spacing
\usepackage{parskip}

% Colors based on GitHub's UI
\usepackage{xcolor}
\definecolor{ghblue}{HTML}{0969da}
\definecolor{ghgray}{HTML}{57606a}
\definecolor{ghlightgray}{HTML}{f6f8fa}
\definecolor{ghborder}{HTML}{d0d7de}

% Hyperlinks
\usepackage{hyperref}
\hypersetup{
    colorlinks=true,
    linkcolor=ghblue,
    urlcolor=ghblue,
    citecolor=ghblue,
    pdfborder={0 0 0}
}

% Make boxes available globally (Fixes the \tcbox Undefined error)
\usepackage{tcolorbox}
\tcbuselibrary{breakable}

% Tell LaTeX to number headers down to Level 4 (####)
\setcounter{secnumdepth}{4}

% Section Headings
\usepackage{titlesec}
\titleformat{\section}{\Huge\bfseries\sffamily}{\thesection}{1em}{}[\vspace{0.2em}\color{ghborder}\titlerule]
\titleformat{\subsection}{\Large\bfseries\sffamily}{\thesubsection}{1em}{}
\titleformat{\subsubsection}{\large\bfseries\sffamily}{\thesubsubsection}{1em}{}

% Fix for Level 4 (####) and Level 5 (#####) headers
\titleformat{\paragraph}{\normalsize\bfseries\sffamily}{\theparagraph}{1em}{}
\titlespacing*{\paragraph}{0pt}{1.5ex}{0.5ex}
\titleformat{\subparagraph}{\normalsize\bfseries\sffamily\color{ghgray}}{\thesubparagraph}{1em}{}
\titlespacing*{\subparagraph}{0pt}{1.5ex}{0.5ex}

% Blockquotes (Grey left border)
\usepackage{mdframed}
\newmdenv[
    topline=false,
    rightline=false,
    bottomline=false,
    leftline=true,
    linewidth=4pt,
    linecolor=ghborder,
    backgroundcolor=white,
    innertopmargin=0pt,
    innerbottommargin=0pt,
    innerrightmargin=0pt,
    fontcolor=ghgray
]{ghquote}
\renewenvironment{quote}{\begin{ghquote}}{\end{ghquote}}

% Force LaTeX to break lines even if it creates large spaces
\sloppy 

% Code Blocks (Pandoc injects highlighting macros here)
$if(highlighting-macros)$
$highlighting-macros$

% Enable automatic line wrapping for long code lines
\usepackage{fvextra}
\RecustomVerbatimEnvironment{Highlighting}{Verbatim}{breaklines,breakanywhere,commandchars=\\\{\}}

% Override Pandoc's default Shaded environment for code blocks
\renewenvironment{Shaded}{
  \vspace{0.5em}
  \begin{tcolorbox}[
    breakable,
    colback=ghlightgray,
    colframe=ghborder,
    boxrule=0.5pt,
    arc=4pt,
    left=5pt, right=5pt, top=5pt, bottom=5pt
  ]
}{
  \end{tcolorbox}
  \vspace{0.5em}
}
$endif$

% Pandoc requires this tightlist macro for lists
\providecommand{\tightlist}{%
  \setlength{\itemsep}{0pt}\setlength{\parskip}{0pt}}

% Image handling
\usepackage{graphicx}
\makeatletter
\def\maxwidth{\ifdim\Gin@nat@width>\linewidth\linewidth\else\Gin@nat@width\fi}
\def\maxheight{\ifdim\Gin@nat@height>\textheight\textheight\else\Gin@nat@height\fi}
\makeatother
\setkeys{Gin}{width=\maxwidth,height=\maxheight,keepaspectratio}

% Table support for Pandoc
\usepackage{longtable,booktabs,array}
\usepackage{calc} % for calculating minipage widths

% Correct order of tables after \paragraph or \header
\usepackage{etoolbox}
\makeatletter
\patchcmd\longtable{\par}{\if@noskipsec\mbox{}\fi\par}{}{}
\makeatother

% Allow footnotes in longtable head/foot
\IfFileExists{footnotehyper.sty}{\usepackage{footnotehyper}}{\usepackage{footnote}}
\makesavenoteenv{longtable}

% Inline code styling (grey background, rounded corners, wrappable)
\let\oldtexttt\texttt
\renewcommand{\texttt}[1]{%
  \allowbreak%
  \tcbox[
    on line, 
    boxsep=0pt, 
    left=3pt, right=3pt, top=1.5pt, bottom=1.5pt, 
    colback=ghlightgray, 
    colframe=ghborder, 
    boxrule=0.5pt, 
    arc=2pt
  ]{\oldtexttt{#1}}%
}

    $if(author)$\\[0.5em] {\large $author$}$endif$
    $if(date)$\\[0.5em] {\large $date$}$endif$
\end{center}
\vspace{1em}
$endif$

$body$

\end{document}
